cmake_minimum_required(VERSION 3.16)
project(youyou_preview C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
if(NOT LVGL_DIR)
  message(FATAL_ERROR "Pass -DLVGL_DIR=/path/to/lvgl-8.4.0")
endif()
set(APP ${CMAKE_CURRENT_LIST_DIR}/../firmware/YouyouSpace)
file(GLOB_RECURSE LV_SOURCES CONFIGURE_DEPENDS ${LVGL_DIR}/src/*.c)
add_library(lvgl STATIC ${LV_SOURCES})
target_include_directories(lvgl PUBLIC ${LVGL_DIR} ${APP})
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
target_link_libraries(lvgl PUBLIC m)
add_executable(youyou_preview native_preview.cpp ${APP}/SceneView.cpp ${APP}/youyou_font.c ${APP}/ArtAssets.c ${APP}/PetAssets.c ${APP}/WalkAssets.c ${APP}/youyou_small.c)
target_include_directories(youyou_preview PRIVATE ${APP})
target_link_libraries(youyou_preview PRIVATE lvgl)
add_executable(game_test game_test.cpp)
target_include_directories(game_test PRIVATE ${APP})
enable_testing()
add_test(NAME game_rules COMMAND game_test)
add_executable(pet_test pet_test.cpp)
target_include_directories(pet_test PRIVATE ${APP})
add_test(NAME pet_and_save COMMAND pet_test)
add_executable(view_test view_test.cpp ${APP}/SceneView.cpp ${APP}/youyou_font.c ${APP}/ArtAssets.c ${APP}/PetAssets.c ${APP}/WalkAssets.c ${APP}/youyou_small.c)
target_include_directories(view_test PRIVATE ${APP})
target_link_libraries(view_test PRIVATE lvgl)
add_test(NAME render_and_bandwidth COMMAND view_test)
add_executable(animation_preview animation_preview.cpp ${APP}/SceneView.cpp ${APP}/youyou_font.c ${APP}/ArtAssets.c ${APP}/PetAssets.c ${APP}/WalkAssets.c ${APP}/youyou_small.c)
target_include_directories(animation_preview PRIVATE ${APP})
target_link_libraries(animation_preview PRIVATE lvgl)
add_executable(city_preview city_preview.cpp ${APP}/SceneView.cpp ${APP}/youyou_font.c ${APP}/ArtAssets.c ${APP}/PetAssets.c ${APP}/WalkAssets.c ${APP}/youyou_small.c)
target_include_directories(city_preview PRIVATE ${APP})
target_link_libraries(city_preview PRIVATE lvgl)
add_executable(city_test city_test.cpp)
target_include_directories(city_test PRIVATE ${APP})
add_test(NAME city_loop_and_input COMMAND city_test)
add_executable(room_preview room_preview.cpp ${APP}/SceneView.cpp ${APP}/youyou_font.c ${APP}/ArtAssets.c ${APP}/PetAssets.c ${APP}/WalkAssets.c ${APP}/youyou_small.c)
target_include_directories(room_preview PRIVATE ${APP})
target_link_libraries(room_preview PRIVATE lvgl)
add_executable(walk_test walk_test.cpp)
target_include_directories(walk_test PRIVATE ${APP})
add_test(NAME pet_walk_and_hitbox COMMAND walk_test)
add_executable(child_test child_test.cpp)
target_include_directories(child_test PRIVATE ${APP})
add_test(NAME child_interactions_and_memories COMMAND child_test)
